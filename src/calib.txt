int _target_speed_l = 0;
int _target_speed_r = 0;

void set_robot_speed(int speed_l, int speed_r){
    _target_speed_l = speed_l;
    _target_speed_r = speed_r;

int last_encoder_read_l = 0;
int last_encoder_read_r = 0;

int get_speed(int motor){
    int delta = 0;
    if (motor == MOTOR_L){
        int current_encoder_read = get_encoder(ENCODER_L);
        delta = get_encoder_diff(last_encoder_read_l, current_encoder_read);
        last_encoder_read_l = current_encoder_read;
    }
    if (motor == MOTOR_R){
        int current_encoder_read = get_encoder(ENCODER_R);
        delta = get_encoder_diff(last_encoder_read_r, current_encoder_read);
        last_encoder_read_r = current_encoder_read;
    }
    int speed = delta;
    return speed;
}
